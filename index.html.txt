            return {
                config(userConfig = {}) {
                    return {
                        content: [],
                        theme: {
                            extend: {
                                colors: {
                                    primary: '#10b981'
                                }
                            }
                        },
                        plugins: [],
                        ...userConfig
                    }
                },
                theme: {
                    extend: {}
                }
            }
        }
        
        // =============================================
        // DATA STORE & PERSISTENCE
        // =============================================
        let state = {
            currentWeek: 1,
            dailyData: {}, // date -> data
            macroSettings: { bodyweight: 185, activity: 1.375, goal: 0 },
            strengthData: {},
            photos: {}, // week -> array of {dataUrl, timestamp, type}
            weeklyReviews: {},
            version: '1.4.2'
        }
        
        const STORAGE_KEY = 'apexCycleTracker_v1'
        
        function loadData() {
            const start = performance.now()
            const saved = localStorage.getItem(STORAGE_KEY)
            if (saved) {
                state = JSON.parse(saved)
            }
            const end = performance.now()
            console.log(`%c[DEBUG] Loaded data in ${(end-start).toFixed(2)}ms`, 'color:#10b981')
            return end - start
        }
        
        function saveData(debounced = true) {
            if (debounced) {
                clearTimeout(window.saveTimeout)
                window.saveTimeout = setTimeout(() => {
                    localStorage.setItem(STORAGE_KEY, JSON.stringify(state))
                    logDebug('Data auto-saved to localStorage')
                }, 800)
            } else {
                localStorage.setItem(STORAGE_KEY, JSON.stringify(state))
                logDebug('Data force-saved')
            }
        }
        
        // =============================================
        // BIBLE VERSES
        // =============================================
        const bibleVerses = [
            { verse: "I can do all things through Christ who strengthens me.", ref: "Philippians 4:13" },
            { verse: "But those who hope in the Lord will renew their strength.", ref: "Isaiah 40:31" },
            { verse: "Be strong and courageous. Do not be afraid; do not be discouraged.", ref: "Joshua 1:9" },
            { verse: "The Lord is my strength and my shield.", ref: "Psalm 28:7" },
            { verse: "My grace is sufficient for you, for my power is made perfect in weakness.", ref: "2 Corinthians 12:9" }
        ]
        
        let currentVerseIndex = 0
        
        function newBibleVerse() {
            currentVerseIndex = (currentVerseIndex + 1) % bibleVerses.length
            document.getElementById('bible-verse').innerHTML = `"${bibleVerses[currentVerseIndex].verse}"`
            document.getElementById('bible-reference').textContent = `— ${bibleVerses[currentVerseIndex].ref}`
            saveData()
        }
        
        // =============================================
        // DAILY SCHEDULE

        const baseSchedule = [

        function renderSchedule() {
            const container = document.getElementById('daily-schedule')
            container.innerHTML = ''
            
            const today = new Date().toISOString().split('T')[0]
            const dayData = state.dailyData[today] || { completed: {}, timestamps: {}, notes: {} }
            
            baseSchedule.forEach(item => {
                // Hide Dbol after week 6
                if (item.dbolOnly && state.currentWeek > 6) return
                
                const isDone = !!dayData.completed[item.id]
                
                const div = document.createElement('div')
                div.className = `flex items-center gap-4 p-4 rounded-3xl ${isDone ? 'bg-emerald-500/10' : 'glass'}`
                div.innerHTML = `
                    <input type="checkbox" ${isDone ? 'checked' : ''} 
                           class="w-6 h-6 accent-emerald-400" 
                           onchange="toggleTask(${item.id}, this.checked)">
                    <div class="flex-1">
                        <div class="flex justify-between">
                            <div>
                                <span class="font-medium">${item.time}</span>
                                <span class="ml-3 text-emerald-400">${item.name}</span>
                            </div>
                            ${item.dose ? `<span class="text-xs text-zinc-400">${item.dose}</span>` : ''}
                        </div>
                    </div>
                    ${isDone ? `<span class="text-[10px] text-emerald-400 font-medium">${dayData.timestamps[item.id] || ''}</span>` : ''}
                `
                container.appendChild(div)
            })
            
            // Update progress ring
            updateProgressRing()
        }
        
        function toggleTask(id, checked) {
            const today = new Date().toISOString().split('T')[0]
            if (!state.dailyData[today]) state.dailyData[today] = { completed: {}, timestamps: {}, notes: {} }
            
            state.dailyData[today].completed[id] = checked
            if (checked) {
                state.dailyData[today].timestamps[id] = new Date().toLocaleTimeString([], {hour:'2-digit', minute:'2-digit'})
            }
            
            renderSchedule()
            saveData()
            
            // Check if day is fully complete
            if (isDayComplete(today)) {
                triggerConfetti()
            }
        }
        
        function isDayComplete(today) {
            const data = state.dailyData[today]
            if (!data) return false
            return baseSchedule.every(item => {
                if (item.dbolOnly && state.currentWeek > 6) return true
                return !!data.completed[item.id]
            })
        }
        
        function updateProgressRing() {
            const today = new Date().toISOString().split('T')[0]
            const data = state.dailyData[today] || { completed: {} }
            
            let total = baseSchedule.length
            if (state.currentWeek > 6) total-- // dbol
            
            const done = Object.keys(data.completed).filter(k => data.completed[k]).length
            const percent = Math.round((done / total) * 100)
            
            const circle = document.getElementById('progress-circle')
            const circumference = 2 * Math.PI * 82
            const offset = circumference - (circumference * percent / 100)
            circle.style.strokeDasharray = circumference
            circle.style.strokeDashoffset = offset
            
            document.getElementById('progress-percentage').textContent = `${percent}%`
            
            // Streak (simple 7-day)
            let streak = 0
            for (let i = 0; i < 7; i++) {
                const d = new Date()
                d.setDate(d.getDate() - i)
                const key = d.toISOString().split('T')[0]
                if (isDayComplete(key)) streak++
                else break
            }
            document.getElementById('streak-count').textContent = streak
        }

        function calculateMacros() {
            const weight = parseFloat(document.getElementById('bodyweight').value) || 185
            const activity = parseFloat(document.getElementById('activity-level').value)
            const goalDelta = parseFloat(document.getElementById('goal').value)
            
            // Harris-Benedict BMR (male assumed)
            const bmr = 88.362 + (13.397 * weight * 0.453592) + (4.799 * 70) - (5.677 * 30) // placeholder height/age
            let tdee = bmr * activity
            const calories = Math.round(tdee + goalDelta)
            
            const protein = Math.round(weight * 1.8)
            const carbs = Math.round(calories * 0.45 / 4)
            const fat = Math.round((calories - (protein * 4) - (carbs * 4)) / 9)
            
            document.getElementById('calories').textContent = calories
            document.getElementById('protein-g').textContent = protein
            document.getElementById('carbs-g').textContent = carbs
            document.getElementById('fat-g').textContent = fat
            
            // Percentages
            const totalCals = calories
            document.getElementById('protein-pct').textContent = Math.round((protein*4/totalCals)*100) + '%'
            document.getElementById('carbs-pct').textContent = Math.round((carbs*4/totalCals)*100) + '%'
            document.getElementById('fat-pct').textContent = Math.round((fat*9/totalCals)*100) + '%'
            
            state.macroSettings = { bodyweight: weight, activity, goal: goalDelta }
            saveData()
            
            renderMacroTracker()
        }

        function handlePhotoUpload(e) {

            const week = state.currentWeek

                setTimeout(() => {

                    confetti.style.left = Math.random() * 100 + 'vw'

                    confetti.style.transform = `rotate(${Math.random()*360}deg)`
                    document.body.appendChild(confetti)
                    
                    const duration = Math.random() * 2500 + 1500
                    confetti.animate([
                        { transform: `translateY(0) rotate(0deg)`, opacity: 1 },
                        { transform: `translateY(${window.innerHeight + 100}px) rotate(${Math.random()*800}deg)`, opacity: 0 }
                    ], { duration, easing: 'cubic-bezier(0.25, 0.1, 0.25, 1)' })
                    
                    setTimeout(() => confetti.remove(), duration)
                }, i * 3)
            }

        function registerServiceWorker() {
            if ('serviceWorker' in navigator) {
                const swScript = `
                    self.addEventListener('install', e => {
                        console.log('%c[Apex] Service Worker installed', 'color:#10b981')
                    })
                    self.addEventListener('fetch', e => {
                        // Cache-first for offline
                        e.respondWith(
                            caches.match(e.request).then(response => {
                                return response || fetch(e.request)
                            })
                        )
                    })
                `
                const blob = new Blob([swScript], {type: 'application/javascript'})
                const swUrl = URL.createObjectURL(blob)
                
                navigator.serviceWorker.register(swUrl).then(reg => {
                    console.log('%c[Apex] PWA Service Worker registered', 'color:#10b981')
                }).catch(err => console.error(err))
            }
        }
